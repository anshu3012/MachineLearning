% One layer of a neural network on one iris flower, written as matrix maths.
% Weights and biases are picked by hand for the illustration: [0.5 0.4] W = [-0.6 -0.3]; + b = [0.6 -0.2]; ReLU = [0.6 0].
\begin{tikzpicture}[every node/.append style={font=\sffamily\Large},
  lab/.style={font=\sffamily\large, text=cgrey, align=center}]
  \node[box=cblue]   (x) at (0,0)    {$\begin{bmatrix} 0.5 & 0.4 \end{bmatrix}$};
  \node[lab, above=2mm of x] {inputs: a vector\\(petal width, sepal width)};
  \node at (2.25,0) {$\times$};
  \node[box=corange] (w) at (4.4,0)  {$\begin{bmatrix} -2 & -1 \\ 1 & 0.5 \end{bmatrix}$};
  \node[lab, above=2mm of w] {weights: a matrix};
  \node at (6.5,0) {$+$};
  \node[box=cgreen]  (b) at (8.6,0)  {$\begin{bmatrix} 1.2 & 0.1 \end{bmatrix}$};
  \node[lab, above=2mm of b] {biases: a vector};
  \node at (10.75,0) {$=$};
  \node[box=cgrey]   (z) at (12.9,0) {$\begin{bmatrix} 0.6 & -0.2 \end{bmatrix}$};
  \node[box=cpurple] (a) at (12.9,-3.2) {$\begin{bmatrix} 0.6 & 0 \end{bmatrix}$};
  \draw[arrow] (z) -- node[right, lab, align=left] {ReLU: every negative\\number becomes 0} (a);
  \node[lab, below=2mm of a] {output of the layer,\\the input of the next layer};
  \node[lab, align=left, anchor=west] at (-1.6,-2.4) {first number:\enspace $0.5 \times (-2) + 0.4 \times 1 + 1.2 = 0.6$\\[2mm]
      second number:\enspace $0.5 \times (-1) + 0.4 \times 0.5 + 0.1 = -0.2$};
\end{tikzpicture}
