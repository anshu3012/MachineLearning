% PDF look for every Note: modelled on the MIT thesis class (mitthesis):
% Latin Modern serif at 12pt, 1in margins, microtype, booktabs tables, bold caption labels.
\usepackage[T1]{fontenc}
\usepackage{lmodern}
\usepackage{microtype}
\usepackage[labelfont=bf]{caption}
\input{glyphtounicode}\pdfgentounicode=1
\pdfinterwordspaceon   % real space characters, so copy/paste keeps spaces
% Boxes. (Not the `framed` package: with a pending figure and a page-splitting table,
% it pushed content off the page and it vanished from the PDF.)
\usepackage[breakable,skins]{tcolorbox}
\definecolor{keybar}{HTML}{4C78A8}
\definecolor{keyfill}{HTML}{EDF2F8}
\definecolor{extrabar}{HTML}{B279A2}
\tcbset{notebox/.style={breakable, enhanced, sharp corners, boxrule=0pt, top=3pt, bottom=3pt, left=8pt, right=4pt,
  before skip=6pt, after skip=8pt}}
% Key point: blue bar on a pale blue background. Extra: purple bar. Plain quote: grey bar.
\newtcolorbox{keypoint}{notebox, colback=keyfill, colframe=keybar, leftrule=4pt}
\newtcolorbox{extra}{notebox, colback=white, colframe=extrabar, leftrule=3pt}
\renewenvironment{quote}{\begin{tcolorbox}[notebox, colback=white, colframe=gray, leftrule=2pt]}{\end{tcolorbox}}
% Figures sit exactly where they are written. Floating figures + page-splitting tables
% pushed text off the page, where it vanished from the PDF.
\usepackage{float}
\floatplacement{figure}{H}
% Contents on its own page.
\let\oldtoc\tableofcontents
\renewcommand{\tableofcontents}{\oldtoc\newpage}
