% Section 3: autoregressive generation. Each word the decoder writes becomes part of its next input.
\begin{tikzpicture}[d/.style={box=corange, minimum width=1.7cm, minimum height=0.8cm, inner sep=3pt, font=\sffamily\small},
  w/.style={font=\sffamily}]
  \foreach \inp/\out [count=\t] in {{<start>}/comment, {comment}/ça, {ça}/va, {va}/{<end>}} {
    \node[d] (d\t) at (3.0*\t, 0) {decoder};
    \node[w, below=0.45cm of d\t] (x\t) {\inp};
    \node[w, above=0.45cm of d\t, text=cgreen] (y\t) {\textbf{\out}};
    \node[note, below=0.15cm of x\t] {step \t};
    \draw[arrow] (x\t) -- (d\t); \draw[arrow] (d\t) -- (y\t);
  }
  \foreach \i/\j in {1/2, 2/3, 3/4} \draw[arrow, draw=cgreen, dashed] (y\i.east) to[out=0, in=180] (x\j.west);
  \node[note, anchor=west] at (13.2, 0) {each output is the next input;\\step $t$ waits for step $t-1$};
\end{tikzpicture}
