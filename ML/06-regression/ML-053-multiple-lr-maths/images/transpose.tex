  \node[below=0.6cm of col.south west, anchor=north west, align=left] (rule) {\large $\beta^{\mathsf T}X^{\mathsf T} = \begin{bmatrix} 1 & 1 \end{bmatrix}\begin{bmatrix} 1 & 1 \\ 2 & 3 \end{bmatrix} = \begin{bmatrix} 1+2 & 1+3 \end{bmatrix} = \begin{bmatrix} \textcolor{cblue}{3} & \textcolor{corange}{4} \end{bmatrix}$};
  \node[note, below=0.2cm of rule.south west, anchor=north west, align=left] {same row: the transpose of a product is the product of the transposes, in reverse order};
