  \node[right=2.2cm of A] (Ai) {\Large $\dfrac{1}{\textcolor{cgreen}{ad - bc}}\begin{bmatrix} \textcolor{cblue}{d} & \textcolor{cred}{-b} \\ \textcolor{cred}{-c} & \textcolor{cblue}{a} \end{bmatrix}$};

  % example
  \node[below=2.4cm of A.south west, anchor=north west] (ex) {\large $\begin{bmatrix} 2 & 5 \\ 5 & 13 \end{bmatrix}^{-1} = \dfrac{1}{2 \times 13 - 5 \times 5}\begin{bmatrix} 13 & -5 \\ -5 & 2 \end{bmatrix} = \begin{bmatrix} 13 & -5 \\ -5 & 2 \end{bmatrix}$};
  \node[note, below=0.15cm of ex.south west, anchor=north west, align=left] {check: $\begin{bmatrix} 2 & 5 \\ 5 & 13 \end{bmatrix}\begin{bmatrix} 13 & -5 \\ -5 & 2 \end{bmatrix} = \begin{bmatrix} 26-25 & -10+10 \\ 65-65 & -25+26 \end{bmatrix} = \begin{bmatrix} 1 & 0 \\ 0 & 1 \end{bmatrix}$, the identity};
