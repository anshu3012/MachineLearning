% Section 7.3: clipping by norm. Example gradient g = (600, 800) has norm 1000 > limit 1,
% so it is scaled by 1/1000 to (0.6, 0.8): same direction, length 1.
\begin{document}
\begin{tikzpicture}[f/.style={font=\large, align=center}]
  \draw[-{Stealth[length=4mm]}, draw=cred, line width=2.4pt] (0, 0) -- (3, 4) node[f, right=4pt, text=cred]
      {gradient $(600, 800)$\\ norm $\sqrt{600^2 + 800^2} = 1000$};
  \draw[-{Stealth[length=4mm]}, draw=cgreen, line width=3pt] (0, 0) -- (0.6, 0.8);
  \draw[cgrey, dashed] (0, 0) circle (1);
  \node[f, text=cgreen, anchor=west] at (1.2, 0.3) {clipped: $(0.6, 0.8)$, norm 1};
  \node[f, text=cgrey, anchor=east] at (-1.1, 0) {limit \texttt{clipnorm=1.0}\\(dashed circle)};
  \node[f] at (1.7, -1.9) {$g \leftarrow g \times \dfrac{1}{1000}$: same direction, safe length};
  \node[note] at (1.7, -2.8) {arrow lengths not to scale: the red one would be 1000 times the green one};
\end{tikzpicture}
\end{document}
