% one column of cells: \col{x}{name}{colour}{v1,v2,...}
\newcommand{\col}[4]{
  \node[font=\sffamily\bfseries\small, text=#3] at (#1,0.6) {#2};
  \foreach \v [count=\i] in {#4} {
    \node[draw=#3, fill=#3!10, minimum width=3.4cm, minimum height=0.75cm] at (#1,-0.75*\i+0.35) {\v};
  }
}
