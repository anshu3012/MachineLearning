    \foreach \i in {1,...,#3} \foreach \j in {1,...,#4}
      \draw[cblue, fill=cblue!25, line width=0.8pt] ({(\j-1)*0.45},{-(\i-1)*0.45}) rectangle ++(0.45,-0.45);
    \node[below=2pt, align=center] at ({#4*0.225},{-#3*0.45}) {#5};
  \end{scope}}
