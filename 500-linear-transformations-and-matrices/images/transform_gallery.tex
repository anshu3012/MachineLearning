% One panel: the grey grid, the moved grid (columns c1, c2), where i-hat (green) and j-hat (red) land.
\newcommand{\panel}[7]{% x, y, title, c1x, c1y, c2x, c2y
  \begin{scope}[xshift=#1cm, yshift=#2cm]
    \clip (-3,-3) rectangle (3,3);
    \draw[cgrey!25, line width=0.4pt] (-3,-3) grid (3,3);
    \begin{scope}[cm={#4,#5,#6,#7,(0,0)}]
      \draw[cblue!55, line width=0.7pt] (-6,-6) grid (6,6);
    \end{scope}
    \draw[cgrey, line width=0.8pt] (-3,0) -- (3,0) (0,-3) -- (0,3);
    \draw[-{Stealth[length=3.5mm]}, cgreen, line width=2.6pt] (0,0) -- (#4,#5);
    \draw[-{Stealth[length=3.5mm]}, cred, line width=2.6pt] (0,0) -- (#6,#7);
  \end{scope}
  \node[font=\bfseries, align=center] at (#1,#2+3.75) {#3};
}
